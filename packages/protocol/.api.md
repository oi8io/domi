# @domi/protocol 事件契约快照

> 由 `bun run scripts/gen-api-snapshot.ts` 生成，**不要手改**。
> 这份文件变了就意味着协议契约变了（INV-01）。
> 改之前先回答一个问题：**用旧版本写下的事件，新代码还能不能解析？**

SCHEMA_VERSION = 6

## 事件类型

ctx.cleanup
  fromSeq: number
  preserved: number[]
  saved: object
  tokensAfter: number
  tokensBefore: number
  toSeq: number

ctx.compact
  fromSeq: number
  keptTurns: number
  summary: object
  tokensAfter: number
  tokensBefore: number
  toSeq: number
  trigger: "threshold" | "manual"

ctx.ref
  fromSeq: number
  sessionId: string
  toSeq: number

error
  counters: object?
  message: string
  recoverable: boolean
  scope: string

fs.snapshot
  bytes: number
  path: string
  phase: "before" | "after"
  sha256: string | null

model.delta
  text: string

model.reason
  text: string

model.request
  model: string
  provider: string
  tokensIn: number

model.switch
  from: string
  lostCapabilities: string[]?
  reason: string?
  to: string

model.usage
  raw: record<string, unknown>

permission
  capabilityId: string
  decision: "allow" | "deny" | "ask"
  matchedRule: string | null
  source: "default" | "config" | "user"

revert
  scope: "files" | "conversation" | "both"
  snapshotId: string | null
  toSeq: number
  undoSnapshotId: string | null

snapshot
  files: number
  id: string
  label: string
  largeFilesSkipped: number?

tool.call
  args: unknown
  id: string
  name: string

tool.result
  id: string
  ms: number
  ok: boolean
  payload: unknown
  reason: string?

user.input
  attachments: object[]?
  text: string
