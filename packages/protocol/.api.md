> 由 `bun run scripts/gen-api-snapshot.ts` 生成，**不要手改**。
> 这份文件变了就意味着协议契约变了（INV-01）。
> 改之前先回答一个问题：**用旧版本写下的事件，新代码还能不能解析？**

SCHEMA_VERSION = 1

## 事件类型

error
  message: string
  recoverable: boolean
  scope: string

model.delta
  text: string

model.reason
  text: string

model.request
  model: string
  provider: string
  tokensIn: number

model.usage
  raw: record

permission
  capabilityId: string
  decision: "allow" | "deny" | "ask"
  matchedRule: string | null
  source: "default" | "config" | "user"

tool.call
  args: unknown
  id: string
  name: string

tool.result
  id: string
  ms: number
  ok: boolean
  payload: unknown
  reason: string?

user.input
  attachments: array?
  text: string
